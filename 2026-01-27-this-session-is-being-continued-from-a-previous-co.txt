  🚀 快速开始（在服务器上）

  # 1. SSH登录到服务器后，克隆代码
  git clone https://github.com/wzh4464/big_matrix_cocluster.git
  cd big_matrix_cocluster

  # 2. 安装依赖
  pip install -e .
  pip install tqdm psutil

  # 3. 使用Shell脚本运行（推荐）
  ./experiment/run_server.sh classic4              # 运行CLASSIC4规模
  ./experiment/run_server.sh large                 # 运行大规模
  ./experiment/run_server.sh batch "classic4 large"  # 批量运行

  # 4. 或使用Python直接运行
  PYTHONPATH=. python experiment/run_large_scale_experiments.py --dataset
  classic4

  # 5. 后台运行（推荐用于长时间实验）
  nohup ./experiment/run_server.sh classic4 > classic4.log 2>&1 &
  tail -f classic4.log  # 查看进度
